cmake_minimum_required(VERSION 3.14)
project(velavision_pool_safety LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(opt LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_CURL
 GGML_BACKEND_DL GGML_OPENMP GGML_BLAS GGML_LLAMAFILE GGML_CUDA GGML_METAL GGML_VULKAN GGML_RPC
 GGML_SYCL GGML_CANN GGML_MUSA GGML_KOMPUTE GGML_OPENCL GGML_ACCELERATE GGML_CPU_KLEIDIAI
 GGML_NATIVE GGML_AVX GGML_AVX2 GGML_BMI2 GGML_FMA GGML_F16C GGML_CCACHE)
 set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp-74d4f5b041ad837153b0e90fc864b8290e01d8d5")
add_subdirectory(${VENDOR} llama)
set_property(TARGET ggml PROPERTY SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/candidate/ggml-backend-reg.cpp")
target_include_directories(ggml PRIVATE ${VENDOR}/ggml/src)
get_target_property(cpu_sources ggml-cpu SOURCES)
list(FILTER cpu_sources EXCLUDE REGEX "(^|/)ggml-cpu[.]c$")
list(APPEND cpu_sources "${CMAKE_CURRENT_SOURCE_DIR}/candidate/ggml-cpu.c")
set_property(TARGET ggml-cpu PROPERTY SOURCES "${cpu_sources}")
target_compile_definitions(ggml-cpu PRIVATE GGML_POOL_POSIX=1)
target_include_directories(ggml-cpu PRIVATE ${VENDOR}/ggml/src/ggml-cpu PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(pool-test tests/pool.c)
set_property(TARGET pool-test PROPERTY C_STANDARD 11)
target_compile_options(pool-test PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(pool-test PRIVATE ggml ggml-cpu)
set_property(TARGET pool-test PROPERTY LINKER_LANGUAGE CXX)
