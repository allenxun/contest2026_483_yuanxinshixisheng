3: # nsync provides portable synchronization primitives, such as mutexes and
4: # condition variables.
5: if ("${CMAKE_MAJOR_VERSION}X" STREQUAL "2X")
6: 	project (nsync)
7: else ()
8: 	# fetch from VERSION file in root
9: 	file (STRINGS VERSION VERSION_STRING)

11: 	cmake_policy(SET CMP0048 NEW) # remove when min ver >=3
12: 
13: 	project (nsync
14: 		VERSION "${VERSION_STRING}")
15: endif ()

23: # -----------------------------------------------------------------
24: # Functions to set common options on targets and files.
25: # Should be called on all targets.
26: function (set_c_target tgtname files)
27: 	if ("${CMAKE_SYSTEM_NAME}X" STREQUAL "LinuxX")
28: 		target_include_directories ("${tgtname}" BEFORE PRIVATE
29: 			"${PROJECT_SOURCE_DIR}/platform/linux"
30: 		)
31: 	endif ()
32: endfunction (set_c_target)
33: 
34: function (set_cpp_target tgtname files)
35: 	target_include_directories ("${tgtname}" BEFORE PRIVATE
36: 		"${PROJECT_SOURCE_DIR}/platform/c++11"
37: 	)
38: 
39: 	if ("${CMAKE_SYSTEM_NAME}X" STREQUAL "LinuxX")
40: 		target_include_directories ("${tgtname}" BEFORE PRIVATE
41: 			"${PROJECT_SOURCE_DIR}/platform/c++11.futex"
42: 		)
43: 	endif ()
44: 
45: 	target_compile_definitions ("${tgtname}" PRIVATE "${NSYNC_CPP_DEFINITIONS}")
46: 
47: 	foreach (s IN ITEMS ${files})
48: 		set_source_files_properties ("${s}"
49: 			PROPERTIES LANGUAGE CXX
50: 			COMPILE_FLAGS "${NSYNC_CPP_FLAGS}")
51: 	endforeach (s)
52: endfunction (set_cpp_target)

266: foreach (s IN ITEMS ${NSYNC_COMMON_SRC} ${NSYNC_OS_CPP_SRC})
267: 	# The C and C++ libraries are built from the same source files but
268: 	# compiled with either the C or C++ compiler. CMake normally detects
269: 	# the filetype based on the extension so we need to override LANGUAGE
270: 	# manually. A custom command first copies the source files (only if
271: 	# changed) so the LANGUAGE property does not conflict.
272: 	add_custom_command (
273: 		OUTPUT cpp/${s}
274: 		COMMAND ${CMAKE_COMMAND} -E copy_if_different ${PROJECT_SOURCE_DIR}/${s} cpp/${s}
275: 		DEPENDS ${PROJECT_SOURCE_DIR}/${s}
276: 	)
277: 
278: 	set (NSYNC_CPP_SRC_COPY ${NSYNC_CPP_SRC_COPY}
279: 		cpp/${s}
280: 	)
281: endforeach (s)

17: /* This module implements a binary semaphore using C++11 constructs.
18: 
19:    The interface of std::condition_variable requires the use of
20:    std::unique_lock, but the justification of correctness here is more subtle
21:    than I'd like.  Presumably, the idea behind std::unique_lock was to provide
22:    exception safety:  it guarantees to release the lock no matter how a block
23:    is exited.  However, if the lock had a non-trivial monitor invariant,
24:    std::unique_lock wouldn't restore it, so the programmer would be forced to
25:    maintain/restore the monitor invariant at every point in the region where an
26:    exception could be thown.  This is why other languages with locks and
27:    exceptions provide try-finally, even if they also provide scoped locked
28:    regions.
29: 
30:    In this code we can get away with std::unique_lock only because the
31:    monitor invariant happens to be trivial, and holds at all procedure call
32:    sites within the region.  The lock and unlock calls are explicit so readers
33:    can see the region bounaries.  */
34:

48: /* Initialize *s; the initial value is 0. */
49: void nsync_mu_semaphore_init (nsync_semaphore *s) {
50: 	struct mutex_cond *mc = (struct mutex_cond *) s;
51: 	new (&mc->mu) std::mutex ();
52: 	new (&mc->cv) std::condition_variable ();
53: 	mc->i = 0;
54: }

56: /* Wait until the count of *s exceeds 0, and decrement it. */
57: void nsync_mu_semaphore_p (nsync_semaphore *s) {
58: 	struct mutex_cond *mc = (struct mutex_cond *) s;
59:         /* std::unique_lock works here because the monitor invariant holds
60:            throughout the region.  */
61: 	std::unique_lock<std::mutex> mc_mu(mc->mu, std::defer_lock);
62: 
63: 	mc_mu.lock ();
64: 	while (mc->i == 0) {
65: 		mc->cv.wait (mc_mu);
66: 	}
67: 	mc->i = 0;
68: 	mc_mu.unlock ();
69: }

72:    the count of *s is non-zero, in which case decrement *s and return 0;

75: 					nsync_time abs_deadline) {
76: 	struct mutex_cond *mc = (struct mutex_cond *)s;
77: 	std::cv_status res = std::cv_status::no_timeout;
78:         /* std::unique_lock works here because the monitor invariant holds
79:            throughout the regions. */
80: 	std::unique_lock<std::mutex> mc_mu(mc->mu, std::defer_lock);
81: 
82: 	if (nsync_time_cmp (abs_deadline, nsync_time_no_deadline) == 0) {
83: 		mc_mu.lock ();
84: 		while (mc->i == 0) {
85: 			mc->cv.wait (mc_mu);
86: 		}
87: 		mc->i = 0;
88: 		mc_mu.unlock ();
89: 	} else {
90: 		nsync_cpp_time_point_ tp_deadline = nsync_to_time_point_ (abs_deadline);
91: 		mc_mu.lock ();
92: 		while (mc->i == 0 &&
93: 		       ((res = mc->cv.wait_until (mc_mu, tp_deadline)) == std::cv_status::no_timeout ||
94: 			(res == std::cv_status::timeout && /* Various systems wake up too early. */
95: 			 nsync_time_cmp (abs_deadline, nsync_time_now ()) > 0))) {
96: 		}
97: 		ASSERT (res == std::cv_status::no_timeout ||
98: 			res == std::cv_status::timeout);
99: 		if (mc->i != 0) {
100: 			res = std::cv_status::no_timeout;
101: 			mc->i = 0;
102: 		}
103: 		mc_mu.unlock ();
104: 	}
105: 	return (res == std::cv_status::timeout? ETIMEDOUT : 0);
106: }

47: /* Return the nsync_time corresponding to tp. */
48: nsync_time nsync_from_time_point_ (nsync_cpp_time_point_ tp) {
49:         /* std::chrono::system_clock::to_time_t() says that "[i]t is implementation
50:            defined whether values are rounded or truncated to the required
51:            precision", which is a little tricky to use.
52:            Assume instead that the epoch for std::chrono::system_clock::time_point is
53:            the same as that for a struct timespec.  This seems to be true on at
54:            least Linux, NetBSD, MacOS, and Win32; and with 64 bits gives
55: 	   us an implementation that will work for over 200 years.   By then,
56: 	   we'll be able to call C++17's timespec_get.  */
57:         int64_t ns = std::chrono::duration_cast<std::chrono::nanoseconds>(
58:                 tp.time_since_epoch ()).count();
59: 	struct timespec ts;
60: 	memset (&ts, 0, sizeof (ts));
61: 	ts.tv_sec = ns / NSYNC_NS_IN_S_;
62: 	ts.tv_nsec = (long) (ns - ts.tv_sec * NSYNC_NS_IN_S_);
63: 	return (ts);
64: }

67: nsync_cpp_time_point_ nsync_to_time_point_ (nsync_time t) {
68: 	nsync_cpp_time_point_ tp;
69: 	std::chrono::nanoseconds t_ns(NSYNC_TIME_NSEC (t) +
70:                                       NSYNC_NS_IN_S_ * (int64_t) NSYNC_TIME_SEC (t));
71: 	nsync_cpp_time_point_::duration tp_dur =
72: 		std::chrono::duration_cast<nsync_cpp_time_point_::duration>(t_ns);
73: 	return (tp + tp_dur);

80: nsync_time nsync_time_sleep (nsync_time delay) {
81: 	nsync_time start = nsync_time_now ();
82: 	nsync_time expected_end = nsync_time_add (start, delay);
83: 	nsync_time remain;
84: 	std::chrono::nanoseconds delay_ns(NSYNC_TIME_NSEC (delay) +
85: 					  NSYNC_NS_IN_S_ * (int64_t) NSYNC_TIME_SEC (delay));
86: 	std::this_thread::sleep_for (delay_ns);
87: 	nsync_time actual_end = nsync_time_now ();
88: 	if (nsync_time_cmp (actual_end, expected_end) < 0) {
89: 		remain = nsync_time_sub (expected_end, actual_end);
90: 	} else {
91: 		remain = nsync_time_zero;
92: 	}
93: 	return (remain);
94: }

96: nsync_time nsync_time_add (nsync_time a, nsync_time b) {
97: 	a.tv_sec += b.tv_sec;
98: 	a.tv_nsec += b.tv_nsec;
99: 	if (a.tv_nsec >= NSYNC_NS_IN_S_) {
100: 		a.tv_nsec -= NSYNC_NS_IN_S_;
101: 		a.tv_sec++;
102: 	}
103: 	return (a);
104: }
105: 
106: nsync_time nsync_time_sub (nsync_time a, nsync_time b) {
107: 	a.tv_sec -= b.tv_sec;
108: 	if (a.tv_nsec < b.tv_nsec) {
109: 		a.tv_nsec += NSYNC_NS_IN_S_;
110: 		a.tv_sec--;
111: 	}
112: 	a.tv_nsec -= b.tv_nsec;
113: 	return (a);
114: }

116: int nsync_time_cmp (nsync_time a, nsync_time b) {
117: 	int cmp = (NSYNC_TIME_SEC (a) > NSYNC_TIME_SEC (b)) -
118: 		  (NSYNC_TIME_SEC (a) < NSYNC_TIME_SEC (b));
119: 	if (cmp == 0) {
120: 		cmp = (NSYNC_TIME_NSEC (a) > NSYNC_TIME_NSEC (b)) -
121: 		      (NSYNC_TIME_NSEC (a) < NSYNC_TIME_NSEC (b));
122: 	}
123: 	return (cmp);
124: }

37: #define FUTEX_WAIT_BITS_ FUTEX_BITSET_MATCH_ANY

40: #define FUTEX_WAIT_BITS_ 0
41: #endif
42: #define FUTEX_WAKE_ (FUTEX_WAKE | FUTEX_PRIVATE_FLAG_)
43: #define FUTEX_TIMEOUT_IS_ABSOLUTE (FUTEX_WAIT_BITS_ != 0)

45: #define ASSERT(x) do { if (!(x)) { *(volatile int *)0 = 0; } } while (0)

47: struct futex {
48: 	int i;  /* lo half=count; hi half=waiter count */
49: };
50: 
51: static nsync_semaphore *sem_big_enough_for_futex = (nsync_semaphore *) (uintptr_t)(1 /
52: 	(sizeof (struct futex) <= sizeof (*sem_big_enough_for_futex)));

54: /* Initialize *s; the initial value is 0. */
55: void nsync_mu_semaphore_init (nsync_semaphore *s) {
56: 	struct futex *f = (struct futex *) s;
57: 	f->i = 0;
58: }
59: 
60: /* Wait until the count of *s exceeds 0, and decrement it. */
61: void nsync_mu_semaphore_p (nsync_semaphore *s) {
62: 	struct futex *f = (struct futex *) s;
63: 	int i;
64: 	do {
65: 		i = ATM_LOAD ((nsync_atomic_uint32_ *) &f->i);
66: 		if (i == 0) {
67:                         int futex_result = futex (&f->i, FUTEX_WAIT_, i, NULL,
68:                                                   NULL, FUTEX_WAIT_BITS_);
69: 			ASSERT (futex_result == 0 || errno == EINTR ||
70: 				errno == EWOULDBLOCK);
71: 		}
72: 	} while (i == 0 || !ATM_CAS_ACQ ((nsync_atomic_uint32_ *) &f->i, i, i-1));
73: }
74: 
75: /* Wait until one of:
76:    the count of *s is non-zero, in which case decrement *s and return 0;
77:    or abs_deadline expires, in which case return ETIMEDOUT. */
78: int nsync_mu_semaphore_p_with_deadline (nsync_semaphore *s, nsync_time abs_deadline) {
79: 	struct futex *f = (struct futex *)s;
80: 	int i;
81: 	int result = 0;
82: 	do {
83: 		i = ATM_LOAD ((nsync_atomic_uint32_ *) &f->i);
84: 		if (i == 0) {
85: 			int futex_result;
86: 			struct timespec ts_buf;
87: 			const struct timespec *ts = NULL;
88: 			if (nsync_time_cmp (abs_deadline, nsync_time_no_deadline) != 0) {
89: 				memset (&ts_buf, 0, sizeof (ts_buf));
90: 				if (FUTEX_TIMEOUT_IS_ABSOLUTE) {
91: 					ts_buf.tv_sec = NSYNC_TIME_SEC (abs_deadline);
92: 					ts_buf.tv_nsec = NSYNC_TIME_NSEC (abs_deadline);
93: 				} else {
94: 					nsync_time now;
95: 					now = nsync_time_now ();
96: 					if (nsync_time_cmp (now, abs_deadline) > 0) {
97: 						ts_buf.tv_sec = 0;
98: 						ts_buf.tv_nsec = 0;
99: 					} else {
100: 						nsync_time rel_deadline;
101: 						rel_deadline = nsync_time_sub (abs_deadline, now);
102: 						ts_buf.tv_sec = NSYNC_TIME_SEC (rel_deadline);
103: 						ts_buf.tv_nsec = NSYNC_TIME_NSEC (rel_deadline);
104: 					}
105: 				}
106: 				ts = &ts_buf;
107: 			}
108: 			futex_result = futex (&f->i, FUTEX_WAIT_, i, ts, NULL, FUTEX_WAIT_BITS_);
109: 			ASSERT (futex_result == 0 || errno == EINTR || errno == EWOULDBLOCK ||
110: 				errno == ETIMEDOUT);
111: 			/* Some systems don't wait as long as they are told. */ 
112: 			if (futex_result == -1 && errno == ETIMEDOUT &&
113: 			    nsync_time_cmp (abs_deadline, nsync_time_now ()) <= 0) {
114: 				result = ETIMEDOUT;
115: 			}
116: 		}
117: 	} while (result == 0 && (i == 0 || !ATM_CAS_ACQ ((nsync_atomic_uint32_ *) &f->i, i, i - 1)));
118: 	return (result);
119: }
120: 
121: /* Ensure that the count of *s is at least 1. */
122: void nsync_mu_semaphore_v (nsync_semaphore *s) {
123: 	struct futex *f = (struct futex *) s;
124:         uint32_t old_value;
125:         do {    
126:                 old_value = ATM_LOAD ((nsync_atomic_uint32_ *) &f->i);
127:         } while (!ATM_CAS_REL ((nsync_atomic_uint32_ *) &f->i, old_value, old_value+1));
128: 	ASSERT (futex (&f->i, FUTEX_WAKE_, 1, NULL, NULL, 0) >= 0);
129: }

21: struct mutex_cond {
22: 	pthread_mutex_t mu;
23: 	pthread_cond_t cv;
24: 	uint32_t i;
25: };
26: 
27: static nsync_semaphore *sem_big_enough_for_mutex_cond =
28: 	(nsync_semaphore *) (1 /
29: 	(sizeof (struct mutex_cond) <= sizeof (*sem_big_enough_for_mutex_cond)));
30: 
31: /* Initialize *s; the initial value is 0. */
32: void nsync_mu_semaphore_init (nsync_semaphore *s) {
33: 	struct mutex_cond *mc = (struct mutex_cond *) s;
34: 	ASSERT (pthread_mutex_init (&mc->mu, NULL) == 0);
35: 	ASSERT (pthread_cond_init (&mc->cv, NULL) == 0);
36: 	mc->i = 0;
37: }
38: 
39: /* Wait until the count of *s exceeds 0, and decrement it. */
40: void nsync_mu_semaphore_p (nsync_semaphore *s) {
41: 	struct mutex_cond *mc = (struct mutex_cond *) s;
42: 	ASSERT (pthread_mutex_lock (&mc->mu) == 0);
43: 	while (mc->i == 0) {
44: 		ASSERT (pthread_cond_wait (&mc->cv, &mc->mu) == 0);
45: 	}
46: 	mc->i = 0;
47: 	ASSERT (pthread_mutex_unlock (&mc->mu) == 0);
48: }
49: 
50: /* Wait until one of:
51:    the count of *s is non-zero, in which case decrement *s and return 0;
52:    or abs_deadline expires, in which case return ETIMEDOUT. */
53: int nsync_mu_semaphore_p_with_deadline (nsync_semaphore *s,
54: 					nsync_time abs_deadline) {
55: 	struct mutex_cond *mc = (struct mutex_cond *)s;
56: 	int res = 0;
57: 	ASSERT (pthread_mutex_lock (&mc->mu) == 0);
58: 	if (nsync_time_cmp (abs_deadline, nsync_time_no_deadline) == 0) {
59: 		while (mc->i == 0) {
60: 			ASSERT (pthread_cond_wait (&mc->cv, &mc->mu) == 0);
61: 		}
62: 		mc->i = 0;
63: 	} else {
64: 		struct timespec ts_deadline;
65: 		memset (&ts_deadline, 0, sizeof (ts_deadline));
66: 		ts_deadline.tv_sec = NSYNC_TIME_SEC (abs_deadline);
67: 		ts_deadline.tv_nsec = NSYNC_TIME_NSEC (abs_deadline);
68: 		while (mc->i == 0 &&
69: 		       ((res = pthread_cond_timedwait (&mc->cv, &mc->mu, &ts_deadline)) == 0 ||
70: 			(res == ETIMEDOUT && /* Various systems wake up too early. */

77: 		}
78: 	}
79: 	ASSERT (pthread_mutex_unlock (&mc->mu) == 0);
80: 	return (res);
81: }

86: 	ASSERT (pthread_mutex_lock (&mc->mu) == 0);
87: 	mc->i = 1;

22:    invoked on int values, because futexes operate on int values.  A
23:    compile-time check in the futex code ensures that both int and   
24:    nsync_atomic_uint32_ are 32 bits.
25: 
26:    Memory barriers:
27: 	   Operations with the suffixes _ACQ and _RELACQ ensure that the operation
28: 	   appears to complete before other memory operations subsequently performed by
29: 	   the same thread, as seen by other threads.  (In the case of ATM_CAS_ACQ,
30: 	   this applies only if the operation returns a non-zero value.)
31: 
32: 	   Operations with the suffixes _REL and _RELACQ ensure that the operation
33: 	   appears to complete after other memory operations previously performed by
34: 	   the same thread, as seen by other threads.  (In the case of ATM_CAS_REL,
35: 	   this applies only if the operation returns a non-zero value.)

37:    // Atomically,
38:    //   int ATM_CAS (nsync_atomic_uint32_ *p, uint32_t old_value, uint32_t new_value) {
39:    //		if (*p == old_value) {
40:    //			*p = new_value;
41:    //			return (some-non-zero-value);
42:    //		} else {
43:    //			return (0);
44:    //		}
45:    //	}
46:    // *_ACQ, *_REL, *_RELACQ variants are available,
47:    // with the barrier semantics described above.

49: 
50:    // Atomically,
51:    //     uint32_t ATM_LOAD (nsync_atomic_uint32_ *p) { return (*p); }
52:    // A *_ACQ variant is available,
53:    // with the barrier semantics described above.

56:    // Atomically,
57:    //     void ATM_STORE (nsync_atomic_uint32_ *p, uint32_t value) { *p = value; }

41: 	HANDLE *h = (HANDLE *) s;

51: 				result = WaitForSingleObject (*h, 0);
52: 			} else {

54: 				delay = nsync_time_sub (abs_deadline, now);

20: int nsync_pthread_cond_timedwait (pthread_cond_t *cv, pthread_mutex_t *mu,
21: 			    const struct timespec *abs_deadline) {
22: 	int result;
23:         if (abs_deadline == NULL) {
24:                 result = SleepConditionVariableCS (cv, mu, INFINITE);
25: 	} else {
26: 		int again;
27: 		do {
28: 			struct timespec now;
29: 			again = 0;
30: 			clock_gettime (CLOCK_REALTIME, &now);
31: 			if (abs_deadline->tv_sec < now.tv_sec ||
32:                             (abs_deadline->tv_sec == now.tv_sec && abs_deadline->tv_nsec <= now.tv_nsec)) {
33: 				result = SleepConditionVariableCS (cv, mu, 0);
34: 			} else {
35: 				struct timespec delay;
36: 				delay = *abs_deadline;
37: 				delay.tv_sec -= now.tv_sec;
38: 				if (delay.tv_nsec < now.tv_nsec) {
39: 					delay.tv_nsec += 1000 * 1000 * 1000;
40: 					delay.tv_sec--;
41: 				}
42: 				delay.tv_nsec -= now.tv_nsec;
43: 				if (delay.tv_sec > 1000*1000) {
44: 					result = SleepConditionVariableCS (cv, mu, 1000*1000);
45: 					again = (result == ERROR_TIMEOUT);
46: 				} else {
47: 					result = SleepConditionVariableCS (cv, mu,
48: 						(unsigned) (delay.tv_sec * 1000 +
49: 						(delay.tv_nsec + 999999) / (1000 * 1000)));
50: 					if (result == ERROR_TIMEOUT) {
51: 						/* Windows often generates early wakeups. */
52: 						clock_gettime (CLOCK_REALTIME, &now);
53: 						again = (abs_deadline->tv_sec > now.tv_sec ||
54: 						         (abs_deadline->tv_sec == now.tv_sec &&
55: 							  abs_deadline->tv_nsec > now.tv_nsec));
56: 					}
57: 				}
58: 			}
59: 		} while (again);
60: 	}
61: 	return (result == ERROR_TIMEOUT? ETIMEDOUT : 0);
62: }

15: #include "headers.h"

17: // Struct representing an element of the "dest" array, whose elements
18: // indicate the destructor asscoaied with the corresponding key.
19: namespace {
20: struct KeyDestructor {
21: 	bool allocated;         // Whether this key is allocated.
22: 	pthread_key_t key;	// Last used version of key.
23: 	void (*dest) (void *);  // Destructor for the key, or nullptr.
24: };
25: } // anonymous namespace
26: 
27: 
28: static SRWLOCK mu = SRWLOCK_INIT;  // protects dest.
29: static const int kMaxKeyShift = 7;
30: static const int kMaxKeys = 1 << kMaxKeyShift;
31: static const int kKeyMask = kMaxKeys - 1;
32: static KeyDestructor dest[kMaxKeys];  // Destructors for keys.
33: 
34: // Element per key per thread.
35: namespace {
36: struct KeyValue {
37: 	pthread_key_t key;  // last key stored
38: 	void *value;
39: };
40: } // anonymous namespace
41: 
42: // Each thread has a class instance that represents an array of pointers
43: // to client-specified thread-specific data.
44: // The class destructor calls the key destructors when the accociated
45: // thread exits.
46: namespace {
47: 
48: class PerThreadData {
49:  public:
50:   PerThreadData() {
51:   }
52:   ~PerThreadData();
53:   void *value(pthread_key_t key) {
54:     void *result = 0;
55:     int i = key & kKeyMask;
56:     if (this->array_[i].key == key) {
57:       result = this->array_[i].value;
58:     }
59:     return (result);
60:   }
61:   void set_value(pthread_key_t key, void *value) {
62:     int i = key & kKeyMask;
63:     this->array_[i].key = key;
64:     this->array_[i].value = value;
65:   }
66:  private:
67:   KeyValue array_[kMaxKeys];
68: };
69: 
70: PerThreadData::~PerThreadData() {
71: 	bool again = true;
72: 	// Try a few times to clear the thread's array.
73: 	for (int attempt = 0; attempt != 4 && again; attempt++) { 
74: 		again = false;
75: 		// Take a snapshot of the destructor state.
76: 		AcquireSRWLockExclusive (&mu);
77: 		KeyDestructor dest_copy[kMaxKeys];
78: 		memcpy (dest_copy, dest, sizeof (dest_copy));
79: 		ReleaseSRWLockExclusive(&mu);
80: 		// Iterate over the values, calling destructors
81: 		// when both destructor and value are non-nil.
82: 		for (int i = 1; i != kMaxKeys; i++) {
83: 			if (dest_copy[i].allocated) {
84: 				void (*destructor) (void *) = dest_copy[i].dest;
85: 				void *value = this->array_[i].value;
86: 				this->array_[i].value = nullptr;
87: 				if (dest_copy[i].key == this->array_[i].key &&
88:                                     destructor != nullptr && value != nullptr) {
89: 					again = true;  // in case destructors set values.
90: 					(*destructor) (value);
91: 				}
92: 			}
93: 		}
94: 	}
95: }
96: } // anonymous namespace
97: 
98: static __declspec(thread) class PerThreadData per_thread_data;
99: 
100: NSYNC_C_START_
101: 
102: int nsync_pthread_key_create (pthread_key_t *pkey, void (*destructor) (void *)) {
103: 	int result = EAGAIN;
104: 	int i;
105: 	AcquireSRWLockExclusive (&mu);
106: 	// Search for an unused slot. Slot 1 is never used.
107: 	for (i = 1; i != kMaxKeys && dest[i].allocated; i++) {
108: 	}
109: 	if (i != kMaxKeys) { // found one
110: 		dest[i].allocated = true;
111: 		dest[i].dest = destructor;
112: 		dest[i].key += kMaxKeys;  // High bits are version number.
113: 		dest[i].key |= i;  // Low bits are slot number.
114: 		*pkey = dest[i].key;
115: 		result = 0;
116: 	}
117: 	ReleaseSRWLockExclusive(&mu);
118: 	return (result);
119: }
120: 
121: int nsync_pthread_key_delete (pthread_key_t key) {
122: 	int result = EINVAL;
123: 	int i = key & kKeyMask;
124: 	AcquireSRWLockExclusive (&mu);
125: 	if (key > 0 && dest[i].allocated) {
126: 		dest[i].allocated = false;
127: 		dest[i].dest = nullptr;
128: 		result = 0;
129: 	}
130: 	ReleaseSRWLockExclusive(&mu);
131: 	return (result);
132: }

134: void *nsync_pthread_getspecific (pthread_key_t key) {
135: 	return (per_thread_data.value (key));
136: }
137: 
138: int nsync_pthread_setspecific (pthread_key_t key, void *value) {
139: 	int result = EINVAL;
140: 	if (key > 0) {
141: 		per_thread_data.set_value (key, value);
142: 		result = 0;
143: 	}
144: 	return (result);
145: }
