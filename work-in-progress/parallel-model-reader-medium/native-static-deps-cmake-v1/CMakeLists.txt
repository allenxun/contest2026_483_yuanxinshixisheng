cmake_minimum_required(VERSION 3.26)
# Supply the real NuttX toolchain on the first configure. No platform masquerade.
project(k7_native_static_dependencies LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "NuttX")
  message(FATAL_ERROR "Requires the real NuttX toolchain/CMAKE_SYSTEM_NAME")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "Requires explicit ARM64 target processor")
endif()
set(K7_ABSEIL_SOURCE "" CACHE PATH "Verified Abseil 20240116 source tree")
set(K7_NSYNC_SOURCE "" CACHE PATH "Verified nsync 1.26.0 source tree")
foreach(src IN ITEMS K7_ABSEIL_SOURCE K7_NSYNC_SOURCE)
  if(NOT EXISTS "${${src}}/CMakeLists.txt")
    message(FATAL_ERROR "${src} must identify an existing verified source tree")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static only" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "No dependency tests" FORCE)
set(NSYNC_ENABLE_TESTS OFF CACHE BOOL "No dependency tests" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "No dependency tests" FORCE)
set(ABSL_BUILD_TEST_HELPERS OFF CACHE BOOL "No dependency test helpers" FORCE)
set(ABSL_ENABLE_INSTALL OFF CACHE BOOL "Consume build-tree targets" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "Consistent C++ ABI" FORCE)
# Do not set CMAKE_HAVE_LIBC_PTHREAD / Threads_FOUND / fake Threads::Threads.
# Upstream FindThreads must be satisfied by the actual target toolchain/runtime.
add_subdirectory("${K7_NSYNC_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/nsync" EXCLUDE_FROM_ALL)
add_subdirectory("${K7_ABSEIL_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/abseil" EXCLUDE_FROM_ALL)
if(NOT TARGET nsync::nsync_cpp)
  add_library(nsync::nsync_cpp ALIAS nsync_cpp)
endif()
target_include_directories(nsync_cpp INTERFACE "${K7_NSYNC_SOURCE}/public")
target_compile_definitions(nsync_cpp INTERFACE NSYNC_ATOMIC_CPP11)
include("${CMAKE_CURRENT_LIST_DIR}/ort-abseil-targets.cmake")
add_library(k7_ort_static_deps INTERFACE)
target_link_libraries(k7_ort_static_deps INTERFACE nsync::nsync_cpp ${K7_ORT_ABSEIL_LIBS})
add_custom_target(k7_native_deps)
foreach(dep IN ITEMS nsync_cpp ${K7_ORT_ABSEIL_LIBS})
  if(NOT TARGET "${dep}")
    message(FATAL_ERROR "Missing upstream target ${dep}")
  endif()
  get_target_property(real "${dep}" ALIASED_TARGET)
  if(NOT real)
    set(real "${dep}")
  endif()
  get_target_property(kind "${real}" TYPE)
  if(NOT kind STREQUAL "STATIC_LIBRARY" AND NOT kind STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "Unexpected non-static dependency ${dep}: ${kind}")
  endif()
  add_dependencies(k7_native_deps "${real}")
endforeach()
# Upstream targets supply the real transitive build/link graph. This is NOT a
# standalone link validation and does not manufacture a merged archive.
