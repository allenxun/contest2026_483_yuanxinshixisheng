if(CONFIG_EXAMPLES_K7VOICE)
  include_directories(include src ../k7radio ../k7sound)
  option(K7VOICE_NATIVE_ORT_ADD "Explicit native ORT Add diagnostic" OFF)
  option(K7VOICE_NATIVE_ASR_MODEL "Explicit staged decoder Session diagnostic" OFF)
  option(K7VOICE_NATIVE_ASR_MIC "Explicit recorded microphone recognition" OFF)
  option(K7VOICE_NATIVE_TTS "Offline spoken provisioning with verified ROMFS assets" OFF)
  if(K7VOICE_NATIVE_TTS AND (NOT K7VOICE_NATIVE_ASR_MIC OR NOT CONFIG_FS_ROMFS OR NOT CONFIG_EXAMPLES_K7VOICE_STAGED_ASSETS))
    message(FATAL_ERROR "Spoken provisioning requires microphone ASR and ROMFS resources")
  endif()
  if(K7VOICE_NATIVE_ASR_MIC AND (NOT K7VOICE_NATIVE_ASR_MODEL OR NOT CONFIG_EXAMPLES_K7SOUND))
    message(FATAL_ERROR "Microphone ASR requires model runtime and k7sound")
  endif()
  if(K7VOICE_NATIVE_ASR_MODEL AND NOT K7VOICE_NATIVE_ORT_ADD)
    message(FATAL_ERROR "ASR model diagnostic requires the verified runtime object")
  endif()
  set(k7voice_stack 16384)
  if(K7VOICE_NATIVE_ORT_ADD)
    if(NOT IS_ABSOLUTE "${K7VOICE_ORT_OBJECT}" OR NOT EXISTS "${K7VOICE_ORT_OBJECT}")
      message(FATAL_ERROR "Native ORT diagnostic requires the verified real runtime object")
    endif()
    if(NOT K7VOICE_ORT_OBJECT_SHA256)
      message(FATAL_ERROR "Native ORT object SHA256 is required")
    endif()
    file(SHA256 "${K7VOICE_ORT_OBJECT}" k7voice_actual_ort_sha256)
    if(NOT k7voice_actual_ort_sha256 STREQUAL K7VOICE_ORT_OBJECT_SHA256)
      message(FATAL_ERROR "Native ORT object hash mismatch")
    endif()
    set(k7voice_stack 131072)
  endif()
  nuttx_add_application(NAME k7voice SRCS src/k7voice_main.cpp src/core.cpp
    src/parsers.cpp src/voice_wifi_adapter.cpp src/voice_loop.cpp
    STACKSIZE ${k7voice_stack} PRIORITY 100 MODULE ${CONFIG_EXAMPLES_K7VOICE})
  if(K7VOICE_NATIVE_ORT_ADD)
    target_compile_definitions(apps_k7voice PRIVATE K7VOICE_NATIVE_ORT_ADD=1)
    target_sources(apps_k7voice PRIVATE "${K7VOICE_ORT_OBJECT}")
    # The verified relocatable object is built from src/native_asr_runtime.cpp
    # plus the locked sherpa/ORT runtime and TLS support. Keep it outside the
    # ordinary source list to avoid linking a second runtime copy.
    if(K7VOICE_NATIVE_ASR_MODEL)
      target_compile_definitions(apps_k7voice PRIVATE K7VOICE_NATIVE_ASR_MODEL=1)
    endif()
    if(K7VOICE_NATIVE_ASR_MIC)
      target_compile_definitions(apps_k7voice PRIVATE K7VOICE_NATIVE_ASR_MIC=1)
    endif()
    if(K7VOICE_NATIVE_TTS)
      if(NOT IS_ABSOLUTE "${K7VOICE_TTS_ASSET_INCLUDE}" OR NOT EXISTS "${K7VOICE_TTS_ASSET_INCLUDE}/k7tts_assets_generated.h")
        message(FATAL_ERROR "TTS requires a generated, package-bound RAM asset descriptor")
      endif()
      target_include_directories(apps_k7voice PRIVATE "${K7VOICE_TTS_ASSET_INCLUDE}")
      target_compile_definitions(apps_k7voice PRIVATE K7VOICE_NATIVE_TTS=1)
    endif()
  endif()
endif()
